    import io

    def get_data_row_count_and_trailer_nr(gzipped_file_path):
        with gzip.open(gzipped_file_path, 'rt', encoding='utf-8') as gz_file:
            content = gz_file.read()

        if len(lines) < 2: # At least header and trailer

        data_rows_count = len(lines) - 2 # Header + Trailer
        
        trailer_line = lines[-1]

            # This needs to be dynamic to test the pattern
            generated_timestamp = datetime.now().strftime('%Y%m%d%H%M%S')
            xcom_pushed_filename = f"DWHM_APT_RABATTREPORT_{generated_timestamp}.csv.gz"
