In such cases, compare the files line by line, ignoring only the dynamic timestamp fields. The record count in the footer must match.

```python
# Example pytest for comparing decompressed files
import gzip
import filecmp
import os

def compare_gzipped_csv_files(legacy_file_path, migrated_file_path, ignore_dynamic_footer_fields=None):
    """
    Compares two gzipped CSV files. If ignore_dynamic_footer_fields is provided,
    it's a list of 0-indexed positions in the footer line to ignore during comparison.
    """
    def decompress_and_read_lines(filepath):
        with gzip.open(filepath, 'rt', encoding='utf-8') as f:
            return f.readlines()

    legacy_lines = decompress_and_read_lines(legacy_file_path)
    migrated_lines = decompress_and_read_lines(migrated_file_path)

    assert len(legacy_lines) == len(migrated_lines), \
        f"Row count mismatch: Legacy has {len(legacy_lines)} lines, Migrated has {len(migrated_lines)}."

    for i, (legacy_line, migrated_line) in enumerate(zip(legacy_lines, migrated_lines)):
        if i == len(legacy_lines) - 1 and ignore_dynamic_footer_fields: # Assuming footer is the last line
            legacy_parts = legacy_line.strip().split('|')
            migrated_parts = migrated_line.strip().split('|')

            assert len(legacy_parts) == len(migrated_parts), \
                f"Footer part count mismatch at line {i+1}: Legacy has {len(legacy_parts)}, Migrated has {len(migrated_parts)}."

            # Compare parts, ignoring specified indices
            for j in range(len(legacy_parts)):
                if j not in ignore_dynamic_footer_fields:
                    assert legacy_parts[j] == migrated_parts[j], \
                        f"Footer content mismatch at line {i+1}, part {j+1}: Legacy '{legacy_parts[j]}', Migrated '{migrated_parts[j]}'."
        else:
            assert legacy_line == migrated_line, \
                f"Content mismatch at line {i+1}:\nLegacy: '{legacy_line.strip()}'\nMigrated: '{migrated_line.strip()}'"
    return True

def test_exis_nna_data_output_parity():
    legacy_file = "baseline/DWHM_APT_NNA_Daten_20231031.csv.gz"
    migrated_file = "migrated_output/DWHM_APT_NNA_Daten_20231031.csv.gz"
    # Footer format: X|<DESTINATION_FILE>|<FROM YYYYMMDD>|NR|V_F_NNA_Daten|<SYSDATE YYYYMMDD>
    # The last field (SYSDATE YYYYMMDD) might differ if generated at different times. It's index 5.
    assert compare_gzipped_csv_files(legacy_file, migrated_file, ignore_dynamic_footer_fields=[5])
```

### 6. Data Quality & Row Count Assertions (NNA Data Example)

**Purpose:** To verify basic data quality aspects of the generated output file, specifically row count and schema consistency.

**Setup:**
1.  **Legacy Baseline:** Obtain the row count and header (schema) from a legacy run's output file for a specific `MONATS_ID` (e.g., `202310`).
2.  **Migrated Job:** Run the `exis_nna_data_export_dag` for the same `MONATS_ID` (`20231031`).
3.  **Retrieve Output:** Get the gzipped CSV file from GCS or SFTP.

**Action:**
1.  Decompress the migrated output file.
2.  Read the header line to get column names.
3.  Count the total number of data rows (excluding header and footer).
4.  Parse the footer line to extract the reported record count.

**Pass/Fail Criterion:**
*   The number of columns in the header of the migrated output must match the legacy output.
*   The column names in the header must match the legacy output.
*   The actual number of data rows in the decompressed file must match the record count reported in the footer.
*   The actual number of data rows must match the row count from the legacy baseline output.
*   No critical columns (e.g., `RAHMENVERTRAG`, `MSISDN`) should contain unexpected `NULL` values if they were always populated in the legacy system.

```python
def assert_csv_data_quality(file_path, expected_header, expected_data_row_count, expected_footer_record_count):

        lines = f.readlines()

    header_line = lines[0].strip()
    footer_line = lines[-1].strip()
    data_lines = lines[1:-1] # Exclude header and footer

    actual_header = header_line.split(',')

    assert actual_data_row_count == expected_data_row_count, \

    # 3. Footer Record Count Check
    footer_parts = footer_line.split('|')

    actual_footer_record_count = int(footer_parts[3]) # Assuming record count is at index 3

        f"Footer record count mismatch. Expected: {expected_footer_record_count}, Actual: {actual_footer_record_count}"
```
