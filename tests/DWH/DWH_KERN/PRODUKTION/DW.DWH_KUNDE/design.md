from google.cloud import bigquery

def log_start_reconciliation(**context):
    stichtag = context['templates_dict']['stichtag']
    # XML Print Statement
    print(f"Kundenadressabgleich fuer Lauf {stichtag} angestossen")
    # KSH Print Statement

def check_and_log_discrepancies(project_id, dataset, **context):
    client = bigquery.Client(project=project_id)
    
    # Query reconciled discrepancy model populated by Dataform
    query = f"""
        SELECT COUNT(1) as discrepancy_count 
        FROM `{project_id}.{dataset}.d_abgl_kunde_woech` 
        WHERE status = 'ABWEICHUNG'
    """
    query_job = client.query(query)
    results = query_job.result()
    row = next(results)
    count = row.discrepancy_count
    
    # Literal original logging logic
    print(f"Anzahl gefundener Abweichungen: {count}")
    print("Adressabgleich Kundenstammdaten ohne erkennbare Fehler beendet")

### File 3: `definitions/d_abgl_kunde_woech.sqlx`
This model is triggered during the Dataform sequence to compute address inconsistencies.

```sql
config {
  type: "incremental",
  schema: "dw_kunde",
  name: "d_abgl_kunde_woech",
  description: "Reconciled customer address master data highlighting discrepancies"
}

-- Oracle SQL converted to BigQuery SQL dialect inside Dataform
SELECT
  src.kunden_id,
  src.adresse_strasse,
  ref.adresse_strasse AS ref_adresse_strasse,
  src.adresse_plz,
  ref.adresse_plz AS ref_adresse_plz,
  src.adresse_ort,
  ref.adresse_ort AS ref_adresse_ort,
  CURRENT_TIMESTAMP() AS reconciliation_timestamp,
  CASE
    WHEN src.adresse_strasse != ref.adresse_strasse OR src.adresse_plz != ref.adresse_plz OR src.adresse_ort != ref.adresse_ort THEN 'ABWEICHUNG'
    ELSE 'OK'
  END AS status
FROM
  `_SESSION.dw_kunde.kunden_master_stg` src
JOIN
  `_SESSION.dw_kunde.kunden_reference_stg` ref
ON
  src.kunden_id = ref.kunden_id
```

CREATE OR REPLACE PROCEDURE `dw_stage.sp_r_abgl_kunde_woech`(
  IN pi_stichtag STRING,
  OUT po_abweichungen INT64,
  OUT po_status_message STRING
)
BEGIN
  -- Declare local variables
  DECLARE v_stichtag STRING;
  DECLARE v_proc_name STRING DEFAULT 'r_abgl_kunde_woech';
  DECLARE v_log_id STRING;
  DECLARE v_log_ts TIMESTAMP;

  SET v_log_ts = CURRENT_TIMESTAMP();
  SET v_log_id = GENERATE_UUID();

  -- 1. Calculate Default Stichtag if not provided
  IF pi_stichtag IS NULL OR pi_stichtag = '' THEN
    -- Default to 7 days ago (Format: YYYYMMDD)
    SET v_stichtag = FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY));
  ELSE
    SET v_stichtag = pi_stichtag;
  END IF;

  -- Log start of execution
  INSERT INTO `dw_log.execution_logs` (log_id, proc_name, log_timestamp, log_level, message)
  VALUES (v_log_id, v_proc_name, v_log_ts, 'INFO', CONCAT('Starte Adressabgleich Kundenstammdaten fuer Stichtag ', v_stichtag));

  -- 2. Execute Reconciliation Logic (Migration of d_abgl_kunde_woech.sql)
  -- This block simulates the processing of customer master data against the reference system.
  -- Output discrepancies are written directly to a staging table.
  
  CREATE OR REPLACE TEMP TABLE temp_abgl_kunde_results AS
  WITH kundendaten AS (
    -- Simulated source table selection
    SELECT 
      kunde_id,
      name,
      strasse,
      plz,
      ort,
      stichtag
    FROM `dw_core.t_kunde`
    WHERE stichtag = v_stichtag
  ),
  stammdaten AS (
    -- Simulated reference table selection
    SELECT 
      kunde_id,
      name,
      strasse,
      plz,
      ort
    FROM `dw_reference.t_stammdaten`
  )
  SELECT 
    k.kunde_id,
    CASE 
      WHEN k.strasse != s.strasse THEN 'ABWEICHUNG: Strasse weicht ab'
      WHEN k.plz != s.plz THEN 'ABWEICHUNG: PLZ weicht ab'
      WHEN k.ort != s.ort THEN 'ABWEICHUNG: Ort weicht ab'
      ELSE 'OK'
    END AS status_abgleich,
    CURRENT_TIMESTAMP() as validation_ts
  FROM kundendaten k
  INNER JOIN stammdaten s ON k.kunde_id = s.kunde_id;

EXCEPTION WHEN ERROR THEN

import logging
from datetime import datetime, timedelta
from airflow import DAG
from airflow.operators.python import PythonOperator
from airflow.models import Variable
from google.cloud import bigquery

# Default DAG configuration
default_args = {
    'owner': 'DWH_KUNDE_OWNER',
    'depends_on_past': False,
    'email_on_failure': True,
    'retries': 1,
    'retry_delay': timedelta(minutes=10),
}

# Fetch Global and Job-Specific Configs from Airflow Variables
GCP_PROJECT = Variable.get("GCP_PROJECT")
BQ_DATASET_DWH = Variable.get("BQ_DATASET_DWH", default_var="DWH_KERN")
BQ_DATASET_STAMM = Variable.get("BQ_DATASET_STAMM", default_var="STAMMDATEN")

def run_weekly_reconciliation(**context):
    # Standard runtime execution parameter
    stichtag = context['templates_dict']['stichtag']
    
    # 1. XML Original Print Message
    # RULE: German literal preserved verbatim
    logging.info(f"Kundenadressabgleich fuer Lauf {stichtag} angestossen")
    
    # 2. KSH Original Startup Message
    # RULE: German literal preserved verbatim
    logging.info(f"Starte Adressabgleich Kundenstammdaten fuer Stichtag {stichtag} ...")

    SELECT
      'ABWEICHUNG' AS MARKER,
      k.KUNDE,
      k.NACHNAME,
      k.VORNAME,
      k.PLZ,
      k.ORT,
      k.STRASSE,
      r.PLZ       AS REF_PLZ,
      r.ORT       AS REF_ORT,
      r.STRASSE   AS REF_STRASSE
    FROM `{GCP_PROJECT}.{BQ_DATASET_DWH}.T_KUNDE` k
    INNER JOIN `{GCP_PROJECT}.{BQ_DATASET_STAMM}.T_KUNDE_REFERENZ` r
      ON r.KUNDE = k.KUNDE
    WHERE k.AKTUALISIERT_AM <= PARSE_DATE('%Y%m%d', '{stichtag}')
      AND (
            IFNULL(k.PLZ, 'x')     != IFNULL(r.PLZ, 'x')
         OR IFNULL(k.ORT, 'x')     != IFNULL(r.ORT, 'x')
         OR IFNULL(k.STRASSE, 'x') != IFNULL(r.STRASSE, 'x')
          )
    ORDER BY k.KUNDE;
    """
    
    client = bigquery.Client(project=GCP_PROJECT)
    query_job = client.query(query)
    results = query_job.result()
    
    # Convert query results to list to allow count validation (Equivalent of counting rows with "ABWEICHUNG" marker)
    rows = list(results)
    deviation_count = 0
    
    for row in rows:
        if row.MARKER == 'ABWEICHUNG':
            deviation_count += 1
            # Mirror stdout print behavior if discrepancies are found
            logging.info(f"MARKER: {row.MARKER} | KUNDE: {row.KUNDE} | PLZ: {row.PLZ} vs REF_PLZ: {row.REF_PLZ}")
            
    # 3. KSH Discrepancy Count Output Message

    logging.info(f"Anzahl gefundener Abweichungen: {deviation_count}")

    # RULE: German literal preserved verbatim
    logging.info("Adressabgleich Kundenstammdaten ohne erkennbare Fehler beendet")
