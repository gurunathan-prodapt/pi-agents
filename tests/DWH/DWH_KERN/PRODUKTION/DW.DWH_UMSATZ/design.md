# SECTION 1 — DESIGN DOCUMENT

## 1. Overview
This ETL job orchestrates and executes the monthly consolidation of sales revenue data (`UMSATZ`) across all corporate group companies. In the legacy environment, it uses UC4/Automic schedulers to run an Ab Initio graph (`umsatz_konsolidierung.mp`) via a wrapper shell script (`r_umsatz_konsolidierung_monatlich.ksh`).

The prescribed target architecture is **Cloud Composer (Airflow) + Dataproc Serverless (PySpark)**, which maintains high structural alignment while modernizing the parallel execution paradigm of Ab Initio into a managed Spark context.

import os
import logging
from datetime import datetime
from airflow import DAG
from airflow.models import Variable
from airflow.operators.empty import EmptyOperator
from airflow.providers.google.cloud.operators.dataproc import DataprocSubmitJobOperator

# Configure local module/pipeline logger
logger = logging.getLogger("airflow.task")

# ── Environment Values Configuration (Strict Global Resolution) ──
GCP_PROJECT = os.environ.get("GCP_PROJECT")
GCP_REGION = os.environ.get("GCP_REGION")
DATAPROC_CLUSTER = os.environ.get("DATAPROC_CLUSTER")
GCS_BUCKET = os.environ.get("GCS_BUCKET")

# Fallback mechanism via Airflow Variables if env vars are missing
if not GCP_PROJECT:
    GCP_PROJECT = Variable.get("GCP_PROJECT", default_var=None)
if not GCP_REGION:
    GCP_REGION = Variable.get("GCP_REGION", default_var=None)
if not DATAPROC_CLUSTER:
    DATAPROC_CLUSTER = Variable.get("DATAPROC_CLUSTER", default_var=None)
if not GCS_BUCKET:
    GCS_BUCKET = Variable.get("GCS_BUCKET", default_var=None)

# ── Job Configuration ──
JOB_CONFIG = {
    "dwh_job_kennung": "UMSATZ_KONSOLIDIERUNG_MONATLICH",
    "konzerngesellschaft": "ALL",
    "pyspark_uri": f"gs://{GCS_BUCKET}/dw/dwh_umsatz/pyspark/r_umsatz_konsolidierung_monatlich.py" if GCS_BUCKET else None
}

default_args = {
    "owner": "DWH_Team",
    "depends_on_past": False,
    "start_date": datetime(2026, 1, 1),
    "email_on_failure": False,
    "email_on_retry": False,
    "retries": 0,
}

with DAG(
    dag_id="dw_dwh_umsatz_konsolidierung_monatlich_jp",
    default_args=default_args,
    description="Monatliche Konsolidierung der Umsatzdaten (UMSATZ) ueber alle Konzerngesellschaften",
    schedule="0 3 1 * *",  # Monthly run on the 1st of the month at 03:00 AM
    catchup=False,
    max_active_runs=1,
    is_paused_upon_creation=False,
    tags=["dwh", "umsatz", "konsolidierung", "monthly"],
) as dag:

    start = EmptyOperator(task_id="start")

    # Dynamic execution monat determined using execution logical_date template context
    verarbeitungsmonat = "{{ logical_date.strftime('%Y%m') }}"
    konzerngesellschaft = JOB_CONFIG["konzerngesellschaft"]

    # Target Dataproc PySpark Execution Job definition
    pyspark_job = {
        "reference": {
            "project_id": GCP_PROJECT,
        },
        "placement": {
            "cluster_name": DATAPROC_CLUSTER,
        },
        "pyspark_job": {
            "main_python_file_uri": JOB_CONFIG["pyspark_uri"],
            "args": [
                "--verarbeitungsmonat", verarbeitungsmonat,
                "--konzerngesellschaft", konzerngesellschaft,
                "--job_kennung", JOB_CONFIG["dwh_job_kennung"]
            ],
        },
    }

    # Task equivalent to the UC4 Unix job execution
    run_spark_consolidation = DataprocSubmitJobOperator(
        task_id="dw_dwh_umsatz_konsolidierung_monatlich_js",
        project_id=GCP_PROJECT,
        region=GCP_REGION,
        job=pyspark_job,
        job_id="dw_ums_kons_mon_{{ ts_nodash }}",
    )

    end = EmptyOperator(task_id="end")

    start >> run_spark_consolidation >> end

#!/usr/bin/env python3
import sys
import argparse
import subprocess

def main():
    # Parse CLI Parameters passed by the Airflow orchestration layer
    parser = argparse.ArgumentParser(description="Monatliche Umsatzkonsolidierung Shell Wrapper")
    parser.add_argument("--verarbeitungsmonat", required=True, help="Processing month in YYYYMM format")
    parser.add_argument("--konzerngesellschaft", default="ALL", help="Target company filter")
    parser.add_argument("--job_kennung", default="UMSATZ_KONSOLIDIERUNG_MONATLICH", help="Job signature")
    args = parser.parse_args()

    v_monat = args.verarbeitungsmonat
    v_gesellschaft = args.konzerngesellschaft

    # Log operational start to match legacy log schema exactly
    # !!! RULE CHECK: PRINT/LOG TEXTS ARE RETAINED VERBATIM (Original German) !!!
    print(f'Umsatzkonsolidierung fuer Monat {v_monat}, Konzerngesellschaft {v_gesellschaft} angestossen')

    # Execute the primary Ab Initio converted PySpark job
    try:
        cmd = [
            "spark-submit",
            "/dw/dwh_umsatz/pyspark/umsatz_konsolidierung.py",
            "--verarbeitungsmonat", v_monat,
            "--konzerngesellschaft", v_gesellschaft,
            "--job_kennung", args.job_kennung
        ]
        result = subprocess.run(cmd, capture_output=True, text=True, check=True)
        print(result.stdout)
        
    except subprocess.CalledProcessError as e:
        print(e.stderr)
        # Replicating run abort error messages from r_umsatz_konsolidierung_monatlich.ksh
        print(f"Umsatzkonsolidierung fuer Monat {v_monat} mit Fehlercode {e.returncode} abgebrochen")
        sys.exit(e.returncode)

if __name__ == "__main__":
    main()
